# 1. Navigate to your project root directory
cd D:\benedictmart\benedictmart\jerome_zom

# 2. Clean and build the project with Maven
mvn clean package

# 3. Set the Tomcat home environment variable
$env:CATALINA_HOME="D:\benedictmart\benedictmart\jerome_zom\tomcat"

# 4. Stop Tomcat gracefully
& "$env:CATALINA_HOME\bin\shutdown.bat"
Start-Sleep -Seconds 3

# 5. Clean up old webapp deployments and old war files
Remove-Item -Recurse -Force "$env:CATALINA_HOME\webapps\benedictjeromemart" -ErrorAction SilentlyContinue
Remove-Item -Force "$env:CATALINA_HOME\webapps\benedictjeromemart.war" -ErrorAction SilentlyContinue
Remove-Item -Force "$env:CATALINA_HOME\webapps\Food-Order-System.war" -ErrorAction SilentlyContinue

# 6. Copy the newly built WAR file into Tomcat webapps and rename it to your app path
Copy-Item "D:\benedictmart\benedictmart\jerome_zom\target\benedictjeromemart.war" -Destination "$env:CATALINA_HOME\webapps\benedictjeromemart.war" -Force

# 7. Start Tomcat
& "$env:CATALINA_HOME\bin\startup.bat"
